function ConnectionsTable = interpret_heptagon_data(res, solution_index)
% INTERPRET_HEPTAGON_DATA 解读 C++ 搜索结果并输出连接列表
%
% 输入:
%   res            : C++ mexFunction 返回的结构体 (包含 .count 和 .solutions)
%   solution_index : (可选) 指定查看第几个解，默认为 1
%
% 输出:
%   ConnectionsTable : 包含完整连接信息的 MATLAB Table
%

    % 1. 检查输入
    if nargin < 2
        solution_index = 1;
    end

    if res.count == 0
        fprintf('结果为空 (res.count = 0)，无法解读。\n');
        ConnectionsTable = [];
        return;
    end

    if solution_index > res.count
        error('指定的解索引 (%d) 超出总解数 (%d)。', solution_index, res.count);
    end

    % 2. 获取原始数据矩阵
    % 数据格式: Target_N 行 x 7 列
    % 矩阵中的值 = (邻居ID_1based * 100) + 邻居边ID_0to6
    RawMatrix = res.solutions{solution_index};
    [N, ~] = size(RawMatrix);

    fprintf('=======================================================\n');
    fprintf('           七边形拼接解 %d (共 %d 个块)\n', solution_index, N);
    fprintf('=======================================================\n');
    fprintf('| %-10s | %-10s | %-4s | %-10s | %-10s |\n', '本块ID', '本块边(0-6)', '连接', '邻居ID', '邻居边(0-6)');
    fprintf('|------------|------------|------|------------|------------|\n');

    % 用于存储表格数据的数组
    src_blocks = [];
    src_sides  = [];
    dst_blocks = [];
    dst_sides  = [];

    % 3. 遍历解析
    for i = 1:N
        for side = 0:6
            % MATLAB 列索引从1开始，所以 side+1
            val = RawMatrix(i, side + 1);

            % val = 0 表示该边未连接（空气）
            if val == 0
                continue;
            end

            % 解码逻辑 (对应 C++: (block_id + 1) * 100 + side)
            neighbor_id = floor(double(val) / 100);
            neighbor_side = mod(double(val), 100);

            % 为了避免输出重复信息 (例如 A->B 和 B->A)，我们通常只打印 ID 较小指向较大的，
            % 或者全部打印。
            % 既然要求"明确列表输出"，为了查阅方便，这里输出【每一条】连接。

            fprintf('| %-10d | %-10d | <--> | %-10d | %-10d |\n', ...
                i, side, neighbor_id, neighbor_side);

            % 收集数据用于生成 Table
            src_blocks = [src_blocks; i];
            src_sides  = [src_sides; side];
            dst_blocks = [dst_blocks; neighbor_id];
            dst_sides  = [dst_sides; neighbor_side];
        end
        % 每块之间画个分隔线，方便阅读
        % fprintf('|------------|------------|------|------------|------------|\n');
    end
    fprintf('=======================================================\n');

    % 4. 生成返回值 Table
    ConnectionsTable = table(src_blocks, src_sides, dst_blocks, dst_sides, ...
        'VariableNames', {'Self_ID', 'Self_Side', 'Neighbor_ID', 'Neighbor_Side'});

    % 简单的统计信息
    total_edges = size(src_blocks, 1);
    fprintf('统计: 共使用了 %d 个块，存在 %d 个连接面 (总边数的一半)。\n', N, total_edges/2);
end


function [Matrix_E_Res, Matrix_A_Res] = analyze_heptagon_result(adj_matrix)
    % 输入: adj_matrix 是 res.solutions{k}，大小为 N x 7
    % 输出: 
    %   Matrix_E_Res: 边矩阵 (去重后)
    %   Matrix_A_Res: 角矩阵 (去重后)

    [N, ~] = size(adj_matrix);

    % --- 数据解析函数 ---
    % 结果存储格式为: (BlockID + 1) * 100 + SideIndex
    % SideIndex 0-13 包含了顺逆时针信息
    decode_node = @(val) floor(val / 100);     % 获取块ID (1 ~ N)
    decode_side = @(val) mod(val, 100);        % 获取边的编号 (0 ~ 13)

    % --- 几何计算函数 ---
    % 计算真实的边编号 (0-6)，忽略顺逆时针差异，因为几何形状不变
    get_real_side = @(s) mod(s, 7);

    % 获取某条边的“前一条边” (在块内部相邻)
    % 在七边形中，边 i 的前一条边总是 (i+6)%7
    get_prev_side = @(s) mod(s + 6, 7);

    % 计算角度类型
    % 规则：夹角由两边编号最大值决定 (0和6例外)
    function angle = calc_angle(s1, s2)
        r1 = mod(s1, 7);
        r2 = mod(s2, 7);
        if (r1 == 0 && r2 == 6) || (r1 == 6 && r2 == 0)
            angle = 0; 
        else
            angle = max(r1, r2);
        end
    end

    % =========================================================
    % 1. 还原边矩阵 (Matrix E)
    % =========================================================
    edge_rows = [];
    for i = 1:N
        for j = 1:7 
            val = adj_matrix(i, j); % i块的第 j-1 边
            if val == 0, continue; end

            neighbor_id = decode_node(val);
            neighbor_side_idx = decode_side(val); % 邻居是用哪个边连过来的

            % 为了不重复记录，只处理 id 小连向 id 大的情况
            if i < neighbor_id
                u = get_real_side(j-1);         % 自己的真实边 (0-6)
                v = get_real_side(neighbor_side_idx); % 邻居的真实边 (0-6)

                if u ~= v
                    row = zeros(1, 7);
                    row(min(u,v)+1) = 1;
                    row(max(u,v)+1) = -1;
                    edge_rows = [edge_rows; row];
                end
            end
        end
    end
    if ~isempty(edge_rows)
        Matrix_E_Res = unique(edge_rows, 'rows');
    else
        Matrix_E_Res = [];
    end

    % =========================================================
    % 2. 还原角矩阵 (Matrix A) - 寻找三阶顶点
    % =========================================================
    angle_rows = [];

    % 遍历每一个块 i 的每一个角
    % 一个角由两条相邻边组成：边 side_curr 和 边 side_prev
    for i = 1:N
        for side_curr = 0:6
            side_prev = get_prev_side(side_curr);

            % 1. 看看这两个相邻边分别连着谁
            val_Q = adj_matrix(i, side_curr + 1);       % 连向 Q
            val_K = adj_matrix(i, side_prev + 1);       % 连向 K

            % 如果有任一边没连，说明这个角没闭合
            if val_Q == 0 || val_K == 0, continue; end

            Q = decode_node(val_Q);
            K = decode_node(val_K);

            % 2. 验证 Q 和 K 是否也连着 (形成闭环 P-Q-K)
            % 我们去 Q 的连接表里找，看有没有连向 K 的
            is_closed = false;
            side_Q_to_K = -1; % Q 用哪条边连向 K

            for s = 1:7
                if decode_node(adj_matrix(Q, s)) == K
                    is_closed = true;
                    side_Q_to_K = s - 1; % 找到连接边
                    break;
                end
            end

            % 3. 如果闭合，说明这是一个三阶顶点
            % 为了不重复计算 (例如算过 P-Q-K，就不再算 Q-K-P 和 K-P-Q)，
            % 我们只在 i 是三者中ID最小的时候记录
            if is_closed && i < Q && i < K

                % --- 现在我们知道这三个块共顶点了，开始算角 ---

                % 角1 (在 P 上): 由 P 的 side_curr 和 side_prev 夹成
                angle_P = calc_angle(side_curr, side_prev);

                % 角2 (在 Q 上): 
                % Q 连向 P 的边是 val_Q 中记录的 side (邻居的边)
                side_Q_to_P = decode_side(val_Q);
                % Q 连向 K 的边是刚才找到的 side_Q_to_K
                angle_Q = calc_angle(side_Q_to_P, side_Q_to_K);

                % 角3 (在 K 上):
                % K 连向 P 的边是 val_K 中记录的 side
                side_K_to_P = decode_side(val_K);
                % K 连向 Q 的边？我们需要去 K 的表里找一下连 Q 的边
                side_K_to_Q = -1;
                for s = 1:7
                    if decode_node(adj_matrix(K, s)) == Q
                        side_K_to_Q = s - 1;
                        break;
                    end
                end
                angle_K = calc_angle(side_K_to_P, side_K_to_Q);

                % --- 记录结果 ---
                row = zeros(1, 8);
                % MATLAB 下标从1开始，所以角度值要+1
                if angle_P < 7, row(angle_P+1) = row(angle_P+1) + 1; end
                if angle_Q < 7, row(angle_Q+1) = row(angle_Q+1) + 1; end
                if angle_K < 7, row(angle_K+1) = row(angle_K+1) + 1; end
                row(8) = 2; % 最后一列固定为2

                angle_rows = [angle_rows; row];
            end
        end
    end

    if ~isempty(angle_rows)
        Matrix_A_Res = unique(angle_rows, 'rows');
    else
        Matrix_A_Res = [];
    end
end
